PLAYWRIGHT — FULL FUNCTION LIBRARY
═══════════════════════════════════════

SETUP:
- npm init -y
- npm i -D @playwright/test
- npx playwright install chromium

LAUNCH:
- const { chromium } = require('playwright');
- const browser = await chromium.launch({ headless: false });
- const context = await browser.newContext({ storageState: 'auth.json' });
- const page = await context.newPage();

NAVIGATION:
- await page.goto('https://site.com')
- await page.goBack()
- await page.reload()
- await page.waitForLoadState('networkidle')

CLICK / MOUSE:
- await page.click(sel)
- await page.dblclick(sel)
- await page.click(sel, { button: 'right' })
- await page.hover(sel)
- await page.mouse.move(x, y)
- await page.mouse.click(x, y)
- await page.mouse.down() / up()
- await page.mouse.wheel(0, 500)

TYPE / KEYBOARD:
- await page.fill(sel, 'text')
- await page.type(sel, 'text', { delay: 100 })
- await page.press(sel, 'Enter')
- await page.keyboard.type('hello')
- await page.keyboard.press('Control+A')

READ DATA:
- await page.textContent(sel)
- await page.innerText(sel)
- await page.getAttribute(sel, 'href')
- await page.inputValue(sel)
- await page.$$eval('a', els => els.map(e => e.href))

FORM:
- await page.selectOption(sel, 'value')
- await page.check(sel) / uncheck(sel)
- await page.setInputFiles(sel, 'file.png')

SCREENSHOT:
- await page.screenshot({ path: 'shot.png', fullPage: true })
- await element.screenshot({ path: 'el.png' })

WAIT:
- await page.waitForSelector(sel)
- await page.waitForResponse(r => r.url().includes('/api'))
- await page.waitForTimeout(2000)

NETWORK:
- await page.route('**/api/*', route => route.fulfill({ json: {...} }))
- page.on('response', r => console.log(r.url(), r.status()))

TABS:
- const p2 = await context.newPage()
- await p2.bringToFront()

SESSION:
- await context.storageState({ path: 'auth.json' })

AUTO-ANSWER FLOW
═══════════════════════════════════════

3. If not found → reason / search

```js

const a = await answerFromMemoryOrReason(q);
await page.fill('.answer-input', a);

```
